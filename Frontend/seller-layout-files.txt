function Header() {
  const { cartCount } = useCart();
  const { user, logout } = useAuth();
  const navigate = useNavigate();
  const [search, setSearch] = useState('');
  const [suggestions, setSuggestions] = useState([]);
  const [menuOpen, setMenuOpen] = useState(false);
  const [accountOpen, setAccountOpen] = useState(false);
  const [categoryMenuOpen, setCategoryMenuOpen] = useState(false);
  const searchRef = useRef(null);
  const menuRef = useRef(null);

  useEffect(() => {
    const trimmedSearch = search.trim();
    if (trimmedSearch.length < 2) {
      setSuggestions([]);
      return;
    }

    const timeout = setTimeout(async () => {
      try {
        const products = await fetchProducts({ search: trimmedSearch, limit: 5 });
        setSuggestions((products.products || []).slice(0, 5));
      } catch {
        setSuggestions([]);
      }
    }, 250);

    return () => clearTimeout(timeout);
  }, [search]);

  useEffect(() => {
    const handleClickOutside = (event) => {
      if (searchRef.current && !searchRef.current.contains(event.target)) {
        setSuggestions([]);
      }

      if (menuRef.current && !menuRef.current.contains(event.target)) {
        setCategoryMenuOpen(false);
      }
    };

    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, []);

  const handleLogout = () => {
    logout();
    setAccountOpen(false);
    setMenuOpen(false);
  };

  const handleSearch = (event) => {
    event.preventDefault();
    const nextQuery = search.trim();
    if (!nextQuery) {
      navigate('/shop');
      return;
    }

    navigate(`/shop?search=${encodeURIComponent(nextQuery)}`);
    setSuggestions([]);
    setMenuOpen(false);
  };

  const navItems = [
    { label: 'Home', to: '/' },
    { label: 'Categories', to: '/categories', isDropdown: true },
    { label: 'Contact', to: '/contact', isDropdown: false },
  ];

  return (
    <header className="topbar-shell">
      <div className="topbar-promo" role="region" aria-label="Store announcements">
        <div className="container promo-inner">
          <span>Free Delivery on Selected Orders | Quality IT Products | Trusted Support</span>
          <Link to="/deals">Live Deals: Up to 20% Off Laptops &amp; Accessories</Link>
        </div>
      </div>
      <div className="header-main">
        <div className="container header-inner">
          <Link to="/" className="brand-block" aria-label="Sylva Technologies homepage">
            <span className="brand-mark">S</span>
            <div>
              <div className="brand-name">Sylva Technologies</div>
              <div className="brand-tagline">Empowering Your Digital Life</div>
            </div>
          </Link>

          <div className="header-tools" ref={searchRef}>
            <form className="header-search" onSubmit={handleSearch}>
              <Search size={16} />
              <input
                type="search"
                value={search}
                onChange={(event) => setSearch(event.target.value)}
                placeholder="Search laptops, routers, accessories..."
                aria-label="Search products"
              />
              {search && (
                <button type="button" className="search-clear" aria-label="Clear search" onClick={() => setSearch('')}>
                  Ã—
                </button>
              )}
            </form>

            {suggestions.length > 0 && (
              <div className="search-suggestions">
                {suggestions.map((product) => (
                  <button
                    key={product._id || product.slug}
                    type="button"
                    className="suggestion-item"
                    onClick={() => {
                      setSearch(product.name);
                      navigate(`/product/${product.slug}`);
                      setSuggestions([]);
                    }}
                  >
                    <span>{product.name}</span>
                    <small>{product.brand}</small>
                  </button>
                ))}
              </div>
            )}
          </div>

          <nav className={`consolidated-nav ${menuOpen ? 'open' : ''}`} ref={menuRef} aria-label="Main navigation">
            {navItems.map((item) => {
              if (item.isDropdown) {
                return (
                  <button
                    key={item.label}
                    type="button"
                    className="nav-link nav-button"
                    onClick={() => setCategoryMenuOpen((value) => !value)}
                  >
                    {item.label}
                    <ChevronDown size={14} />
                  </button>
                );
              }

              return <NavLink key={item.to} to={item.to} className={({ isActive }) => (isActive ? 'nav-link active' : 'nav-link')} onClick={() => setMenuOpen(false)}>{item.label}</NavLink>;
            })}

            {categoryMenuOpen && (
              <div className="mega-menu">
                <div className="mega-menu__header">
                  <span>Categories</span>
                  <Link to="/categories" className="view-all-link" onClick={() => setCategoryMenuOpen(false)}>View all categories</Link>
                </div>
                <div className="mega-menu__grid">
                  {categoryGroups.map((group) => (
                    <div key={group.title} className="mega-menu__group">
                      <h4>{group.title}</h4>
                      <div className="mega-menu__list">
                        {group.items.map((item) => <Link key={item.label} to={item.href} onClick={() => setCategoryMenuOpen(false)}>{item.label}</Link>)}
                      </div>
                    </div>
                  ))}
                </div>
              </div>
            )}
          </nav>
          <NavLink
  to="/sell"
  className={({ isActive }) => (isActive ? 'nav-link active' : 'nav-link')}
  onClick={() => setMenuOpen(false)}
>
  Sell
</NavLink>
          

          <div className="header-actions">
            <div className="account-menu-wrap">
              <button
                type="button"
                className="account-menu-trigger"
                onClick={() => setAccountOpen((value) => !value)}
                aria-expanded={accountOpen}
              >
                <User size={16} />
                <span>{user ? (user.name?.split(' ')[0] || 'Account') : 'Account'}</span>
                <ChevronDown size={14} />
              </button>

              {accountOpen && (
                <div className="account-menu">
                  {!user ? (
                    <>
                      <Link to="/login" onClick={() => setAccountOpen(false)}>Sign In</Link>
                      <Link to="/register" onClick={() => setAccountOpen(false)}>Register</Link>
                    </>
                  ) : (
                    <>
                      <Link to="/profile" onClick={() => setAccountOpen(false)}>My Account</Link>
                      <Link to="/profile" onClick={() => setAccountOpen(false)}>My Orders</Link>
                      <Link to="/profile" onClick={() => setAccountOpen(false)}>Wishlist</Link>
                      <Link to="/profile" onClick={() => setAccountOpen(false)}>Profile</Link>
                      {user.role === 'admin' && (
                        <Link to="/admin" onClick={() => setAccountOpen(false)}>Admin Dashboard</Link>
                      )}
                      <button type="button" className="menu-logout" onClick={handleLogout}>Logout</button>
                    </>
                  )}
                </div>
              )}
            </div>

            <Link to="/cart" className="cart-button icon-link" aria-label="View cart">
              <ShoppingCart size={16} />
              <span className="cart-text">Cart</span>
              <span className="cart-count">({cartCount})</span>
            </Link>

            <button className="mobile-menu" aria-label="Open menu" onClick={() => setMenuOpen((value) => !value)}>
              <Menu size={18} />
            </button>
          </div>
        </div>
      </div>

    </header>
  );
}

export default Header;

          {activeSection === 'products' && (
            <SellerProducts />
          )}

          {/* ORDERS */}
          {activeSection === 'orders' && (
            <section className="seller-dashboard-panel">
              <div className="seller-dashboard-panel-heading">
                <div>
                  <span>SALES</span>
                  <h3>Orders</h3>
                </div>
              </div>

              <div className="seller-dashboard-empty">
                <div className="seller-dashboard-empty-icon">
                  <ClipboardList size={26} />
                </div>

                <h4>No orders yet</h4>

                <p>
                  Customer orders will appear here once
                  your store starts selling.
                </p>
              </div>
            </section>
          )}

          {/* SALES */}
          {activeSection === 'sales' && (
            <section className="seller-dashboard-panel">
              <div className="seller-dashboard-panel-heading">
                <div>
                  <span>ANALYTICS</span>
                  <h3>Sales</h3>
                </div>
              </div>

              <div className="seller-dashboard-empty">
                <div className="seller-dashboard-empty-icon">
                  <BarChart3 size={26} />
                </div>

                <h4>Sales Analytics</h4>

                <p>
                  Sales statistics and revenue reports
                  will appear here.
                </p>
              </div>
            </section>
          )}

          {/* CUSTOMERS */}
          {activeSection === 'customers' && (
            <section className="seller-dashboard-panel">
              <div className="seller-dashboard-panel-heading">
                <div>
                  <span>CUSTOMERS</span>
                  <h3>Customers</h3>
                </div>
              </div>

              <div className="seller-dashboard-empty">
                <div className="seller-dashboard-empty-icon">
                  <Users size={26} />
                </div>

                <h4>No customers yet</h4>

                <p>
                  Customer information and activity will
                  appear here after your first sales.
                </p>
              </div>
            </section>
          )}

          {/* VERIFICATION */}
          {activeSection === 'verification' && (
            <section className="seller-dashboard-panel">
              <div className="seller-dashboard-panel-heading">
                <div>
                  <span>ACCOUNT SECURITY</span>
                  <h3>Seller Verification</h3>
                </div>
              </div>

              <div className="seller-dashboard-empty">
                <div className="seller-dashboard-empty-icon">
                  <ShieldCheck size={26} />
                </div>

                <h4>
                  {statusConfig.label}
                </h4>

                <p>
                  {statusConfig.message}
                </p>
              </div>
            </section>
          )}

          {/* SETTINGS */}
          {activeSection === 'settings' && (
            <section className="seller-dashboard-panel">
              <div className="seller-dashboard-panel-heading">
                <div>
                  <span>ACCOUNT</span>
                  <h3>Settings</h3>
                </div>
              </div>

              <div className="seller-dashboard-empty">
                <div className="seller-dashboard-empty-icon">
                  <Settings size={26} />
                </div>

                <h4>Seller Settings</h4>

                <p>
                  Account and seller preferences will be
                  configured here.
                </p>
              </div>
            </section>
          )}

        </main>
      </div>
    </div>
  );
}

export default SellerDashboardPage;
